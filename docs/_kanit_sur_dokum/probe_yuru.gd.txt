extends Node3D
## Oyuncu yürüme sondası: gerçek Player ile bot; v2 sur dilimi (merdiven → iç sur yolu → uca) ve eski LandWalls (merdiven).
var pl: Player

func _bot(etiket: String, wps: Array, sprint: bool, limit := 120.0) -> void:
	pl.frozen = false
	if sprint: Input.action_press("sprint")
	else: Input.action_release("sprint")
	var t := 0.0
	var i := 0
	var iz := 0.0
	var son := pl.global_position
	var yol := 0.0
	var logs: Array = []
	while i < wps.size() and t < limit:
		var wp: Vector3 = wps[i]
		pl.face(Vector3(wp.x, pl.global_position.y + 1.4, wp.z))
		Input.action_press("move_forward")
		await get_tree().physics_frame
		var dt := get_physics_process_delta_time()
		t += dt
		yol += pl.global_position.distance_to(son)
		son = pl.global_position
		var p := pl.global_position
		if Vector2(p.x - wp.x, p.z - wp.z).length() < 0.9 and absf(p.y - wp.y) < 1.2:
			logs.append("wp%d t=%.1f y=%.2f" % [i, t, p.y])
			i += 1
			iz = t
		elif t - iz > 25.0:
			logs.append("TAKILDI wp%d at (%.1f,%.1f,%.1f) t=%.1f" % [i, p.x, p.y, p.z, t])
			break
	Input.action_release("move_forward")
	Input.action_release("sprint")
	print("ISTWALK ", etiket, " sprint=", sprint, " sure=%.1f yol=%.1f son=(%.1f,%.1f,%.1f)" % [t, yol, pl.global_position.x, pl.global_position.y, pl.global_position.z], " ", " | ".join(logs))

func _tp(p: Vector3) -> void:
	pl.global_position = p
	pl.velocity = Vector3.ZERO

func _ready() -> void:
	pl = Player.new()
	add_child(pl)
	pl.show_remote(false)
	# ---- v2 sur dilimi
	for id in ["sur", "mesoteikhion"]:
		for sp in [false, true]:
			var h := Node3D.new()
			add_child(h)
			var L: float = 110.0 if id == "sur" else 190.0
			var o: Dictionary = {"ad": "SurDilimi"} if id == "sur" else IstSurlar.OPTS[id]
			var sd: int = 1453 if id == "sur" else 1000 + id.hash() % 997
			IstSurlar._run(h, L, 1.0, 0, sd, o)
			for i in 6: await get_tree().physics_frame
			var lay := IstSurlar._layout(L, 1.0, sd, o)
			var ix0: float = lay["in_x0"]
			var ix1: float = lay["in_x1"]
			_tp(Vector3(ix0 - 1.5, 0.1, -3.4))
			await _bot("v2_%s_ic_merdiven_ve_yol_+x" % id, [Vector3(ix0 + 1.0, 0.3, -3.4), Vector3(ix1 + 0.7, 12.0, -3.4), Vector3(ix1 + 0.7, 12.0, 0.0), Vector3(L * 0.5 - 2.0, 12.0, 0.0)], sp)
			# aynı yolda geri (-x) uç
			await _bot("v2_%s_ic_yol_-x" % id, [Vector3(-L * 0.5 + 2.0, 12.0, 0.0)], sp)
			# peribolostan dış sur
			var ox0: float = lay["out_x0"]
			var ox1: float = lay["out_x1"]
			_tp(Vector3(ox0 - 1.5, 0.1, 16.6))
			await _bot("v2_%s_dis_merdiven_ve_yol" % id, [Vector3(ox0 + 1.0, 0.3, 16.6), Vector3(ox1 + 0.5, 8.5, 16.6), Vector3(ox1 + 0.5, 8.5, 18.8), Vector3(L * 0.5 - 2.0, 8.5, 18.8)], sp)
			await _bot("v2_%s_dis_yol_-x" % id, [Vector3(-L * 0.5 + 2.0, 8.5, 18.8)], sp)
			# hendek: yakın yakadan dibe, karşıya, uzak yaka rampası
			var ax: float = lay["ramp_a_x"]
			var bx: float = lay["ramp_b_x"]
			var rl: float = lay["ramp_len"]
			_tp(Vector3(ax + 0.5, 0.1, 31.4))
			await _bot("v2_%s_hendek_gecis" % id, [Vector3(ax - rl + 1.0, -5.8, 31.4), Vector3(ax - rl + 1.0, -5.8, 40.0), Vector3(bx + rl - 1.0, -5.8, 48.8), Vector3(bx - 0.5, 0.0, 48.8)], sp)
			h.queue_free()
			for i in 3: await get_tree().physics_frame
	# ---- eski LandWalls: dış sur merdiveni (x=+8, Ladder ölü 8.6 m, 12°); dış sur yolu uçtan uca
	var w := LandWalls.new()
	w.lite = true
	add_child(w)
	for i in 6: await get_tree().physics_frame
	for sp in [false, true]:
		_tp(Vector3(8.0, 0.1, 12.0))
		pl.face(Vector3(8.0, 1.5, 14.0))
		await _bot("eski_dis_sur_merdiven_x8", [Vector3(8.0, 8.0, 15.0)], sp, 40.0)
		await _bot("eski_dis_sur_yol_+x", [Vector3(48.0, 8.0, 15.0)], sp, 60.0)
		await _bot("eski_dis_sur_yol_-x(gedik)", [Vector3(-48.0, 8.0, 15.0)], sp, 60.0)
		# depo -> gedik yürüyüşü (peribolos)
		_tp(LandWalls.DEPOT + Vector3(0, 0.1, 0))
		await _bot("eski_depo_gedik", [LandWalls.BREACH + Vector3(0, 0.3, -2.0)], sp, 40.0)
		# hendek: sol uçta rampa (x=-38 .. -47), z=33.6 / 22.4
		_tp(Vector3(-47.0, 0.1, 22.4))
		await _bot("eski_hendek_gecis", [Vector3(-38.0, -2.8, 22.4), Vector3(-38.0, -2.8, 33.6), Vector3(-47.0, 0.1, 33.6)], sp, 60.0)
	get_tree().quit(0)
