extends Node3D
## v2 sur sondası (başsız): IstSurlar'ı gerçek kurucuyla (k = IstSehir.K_SUR = 1.0) bir ana düğüme kurar, fizik ışınlarıyla
## yürüme yolunu, peribolosu, hendeği ölçer. Çıktı: stdout (ISTV2 ... satırları).

const K := 1.0
var host: Node3D


func _ready() -> void:
	host = Node3D.new()
	add_child(host)
	var floor_ := StaticBody3D.new()   # referans: yok (zemin ölçümü yalnız surun kendi katıları)
	_run.call_deferred()


func _ray(x: float, z: float, y0 := 60.0, y1 := -20.0) -> float:
	var sp := get_viewport().find_world_3d().direct_space_state
	var q := PhysicsRayQueryParameters3D.create(Vector3(x, y0, z), Vector3(x, y1, z))
	var r := sp.intersect_ray(q)
	if r.is_empty():
		return NAN
	return (r["position"] as Vector3).y


func _run() -> void:
	var d := IstSurlar.dims(K)
	print("ISTV2 dims ", JSON.stringify(d))
	var ids := ["sur", "mesoteikhion", "romanos", "kharisios", "pege", "altin_kapi", "pempton", "kerkoporta", "kaligaria", "rhegion", "ksylokerkos"]
	for id: String in ids:
		var L: float
		var o: Dictionary
		var seed_ := 1453
		if id == "sur":
			L = maxf(110.0 * K, 90.0)
			o = {"ad": "SurDilimi"}
		else:
			L = IstSurlar._gate_len(id, K)
			o = IstSurlar.OPTS[id]
			seed_ = 1000 + id.hash() % 997
		var h := Node3D.new()
		h.name = id
		host.add_child(h)
		IstSurlar._run(h, L, K, 0, seed_, o)
		for i in 4:
			await get_tree().physics_frame
		_olc(id, L, o, seed_)
		h.queue_free()
		for i in 2:
			await get_tree().physics_frame
	# gedikli sur dilimi (Mesoteikhion): 190 m, gedik x -60,0,60 (sehir.gd _gedik_x aynı biçim)
	var h2 := Node3D.new()
	host.add_child(h2)
	IstSurlar._run(h2, 190.0, K, 0, 1453, {"ad": "SurDilimi", "gedik": [-60.0, 0.0, 60.0]})
	for i in 4:
		await get_tree().physics_frame
	_olc("sur_gedikli", 190.0, {"gedik": [-60.0, 0.0, 60.0]}, 1453)
	get_tree().quit(0)


func _profil(id: String, etiket: String, z: float, L: float, y_beklenen: float) -> void:
	# x boyunca ışın: yürünebilir (y_beklenen ± 0.35), yüksek (kule/korkuluk), alçak (boşluk) parçaları
	var seg: Array = []
	var cur := ""
	var x0 := -L * 0.5
	var start := x0
	var x := x0
	var n_yur := 0.0
	while x <= L * 0.5:
		var y := _ray(x, z)
		var t := "yok"
		if not is_nan(y):
			if absf(y - y_beklenen) <= 0.35:
				t = "yol"
			elif y > y_beklenen + 0.35:
				t = "yuksek"
			else:
				t = "alcak"
		if t != cur:
			if cur != "":
				seg.append("%s[%.1f..%.1f]" % [cur, start, x - 0.5])
			cur = t
			start = x
		if t == "yol":
			n_yur += 0.5
		x += 0.5
	seg.append("%s[%.1f..%.1f]" % [cur, start, x - 0.5])
	print("ISTV2 ", id, " ", etiket, " z=", z, " yol_m=", n_yur, " / L=", L, " seg=", " ".join(seg))



func _hray(x0: float, x1: float, y: float, z: float) -> bool:
	var sp := get_viewport().find_world_3d().direct_space_state
	var q := PhysicsRayQueryParameters3D.create(Vector3(x0, y, z), Vector3(x1, y, z))
	return not sp.intersect_ray(q).is_empty()

func _zray(x: float, y: float, z0: float, z1: float) -> float:
	var sp := get_viewport().find_world_3d().direct_space_state
	var q := PhysicsRayQueryParameters3D.create(Vector3(x, y, z0), Vector3(x, y, z1))
	var r := sp.intersect_ray(q)
	return NAN if r.is_empty() else (r["position"] as Vector3).z

func _engel(id: String, etiket: String, z: float, a: float, b: float, y: float) -> void:
	var seg: Array = []
	var x := a
	var cur := false
	var st := a
	while x < b:
		var e := _hray(x, x + 0.5, y, z)
		if e != cur:
			if cur: seg.append("[%.1f..%.1f]" % [st, x])
			cur = e
			st = x
		x += 0.5
	if cur: seg.append("[%.1f..%.1f]" % [st, x])
	print("ISTV2 ", id, " ", etiket, " engel(y=", y, ",z=", z, ")=", " ".join(seg))

func _olc(id: String, L: float, o: Dictionary, seed_: int) -> void:
	var lay := IstSurlar._layout(L, K, seed_, o)
	var d: Dictionary = lay["d"]
	var itx: Array = lay["itx"]
	var xs: Array = []
	for t in itx:
		xs.append("%.1f(w%.1f)" % [float(t["x"]), float(t["w"])])
	var otx: Array = lay["otx"]
	var ox: Array = []
	for t in otx:
		ox.append("%.1f" % float(t))
	print("ISTV2 ", id, " L=", L, " ic_kule_x=", " ".join(xs))
	print("ISTV2 ", id, " dis_kule_x=", " ".join(ox))
	var ks: Array = []
	for kk in ["in_x0", "in_x1", "in_tread", "out_x0", "out_x1", "out_tread", "ramp_len", "ramp_a_x", "ramp_b_x", "gw", "gx"]:
		ks.append("%s=%s" % [kk, str(snappedf(float(lay.get(kk, NAN)), 0.01))])
	print("ISTV2 ", id, " ", " ".join(ks))
	var cl := IstSurlar._climbs(L, K, seed_, o)
	for c in cl:
		var a: Vector3 = c[0]
		var b: Vector3 = c[1]
		print("ISTV2 ", id, " climb ", a, " -> ", b, " uzunluk=%.2f yuk=%.2f" % [a.distance_to(b), b.y - a.y])
	var pe := IstSurlar._perches(L, K, seed_, o)
	print("ISTV2 ", id, " perches=", pe)
	# yürüme yolu profilleri
	var HW: float = d["HW"]
	var OH: float = d["OH"]
	_profil(id, "ic_sur_yolu", 0.0, L, HW)
	_profil(id, "ic_sur_yolu_z-1.5", -1.5, L, HW)
	_engel(id, "ic_sur", 0.0, -L * 0.5, L * 0.5, HW + 0.9)
	_engel(id, "dis_sur", (float(d["zo0"]) + float(d["zo1"])) * 0.5, -L * 0.5, L * 0.5, OH + 0.9)
	var xs0 := L * 0.1
	print("ISTV2 ", id, " yol_eni_ic: z_dis=", _zray(xs0, HW + 0.6, -4.0, 0.0), " z_ic=", _zray(xs0, HW + 0.6, 4.0, 0.0), " | dis: ", _zray(xs0, OH + 0.6, float(d["zo0"]) - 0.5, float(d["zo0"]) + 1.3), " ", _zray(xs0, OH + 0.6, float(d["zo1"]) + 0.5, float(d["zo0"]) + 1.3))
	_profil(id, "dis_sur_yolu", (float(d["zo0"]) + float(d["zo1"])) * 0.5, L, OH)
	# peribolos zemini ve hendek: dikey enine kesit x=L*0.1 ve x=0 (kapıysa yan)
	for xs_ in [L * 0.1, 0.0]:
		var kes: Array = []
		var z := -8.0
		while z <= float(d["zm1"]) + 6.0:
			var y := _ray(xs_, z)
			kes.append("%.1f:%s" % [z, "-" if is_nan(y) else "%.2f" % y])
			z += 1.0
		print("ISTV2 ", id, " enkesit x=%.1f " % xs_, " ".join(kes))
	# çarpışma gövdesi sayısı
	var n_body := 0
	var stack: Array = [host.get_node(id) if host.has_node(id) else host]
	while not stack.is_empty():
		var nd: Node = stack.pop_back()
		if nd is CollisionObject3D:
			n_body += 1
		for c in nd.get_children():
			stack.append(c)
	print("ISTV2 ", id, " carpisma_govde=", n_body)
